use ecotech_db;

alter table EMPLEADO drop index CORREO;

alter table EMPLEADO
	modify CORREO varchar(400) not null,
	modify SALARIO varchar(255) not null,
	modify DIRECCION varchar(400) not null,
	add CORREOHASH char(64) null after CORREO;

update EMPLEADO set CORREOHASH = uuid();

alter table EMPLEADO
	modify CORREOHASH char(64) not null,
	add unique (CORREOHASH);
